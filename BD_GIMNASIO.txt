CREATE DATABASE IF NOT EXISTS db_gimnasio
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE db_gimnasio;

-- =========================================================
-- Tabla: usuarios (para el login / autenticacion)
-- =========================================================
DROP TABLE IF EXISTS usuarios;

CREATE TABLE usuarios (
    id       BIGINT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50)  NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,          -- password cifrado con BCryptPasswordEncoder
    rol      VARCHAR(20)  NOT NULL DEFAULT 'ADMIN',
    activo   BOOLEAN      NOT NULL DEFAULT TRUE
);

-- =========================================================
-- Tabla: socios (CRUD principal del proyecto)
-- =========================================================
DROP TABLE IF EXISTS socios;

CREATE TABLE socios (
    id                BIGINT AUTO_INCREMENT PRIMARY KEY,
    nombres           VARCHAR(60)  NOT NULL,
    apellidos         VARCHAR(60)  NOT NULL,
    dni               VARCHAR(8)   NOT NULL UNIQUE,
    email             VARCHAR(80),
    telefono          VARCHAR(15),
    fecha_nacimiento  DATE,
    plan              VARCHAR(20)  NOT NULL,   -- MENSUAL, TRIMESTRAL, SEMESTRAL, ANUAL
    fecha_inscripcion DATE         NOT NULL,
    activo            BOOLEAN      NOT NULL DEFAULT TRUE
);
